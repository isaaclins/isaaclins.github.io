{{/* Stack table: label column left, values right, hairline row separators. One "label | value" per line in the inner content. Both columns are markdownified, so links and emphasis work in either. */}}
{{- range $line := split (trim .Inner "\n\r \t") "\n" -}} {{- $line = trim $line " \t\r" -}} {{- if $line -}} {{- $parts := split $line "|" -}}
{{ trim (index $parts 0) " " | markdownify }}
{{ if gt (len $parts) 1 }}{{ trim (delimit (after 1 $parts) "|") " " | markdownify }}{{ end }}
{{- end -}} {{- end -}}
